# Minimal, standalone requirements for the inference_app (Streamlit RAG chat).
#
# Install into a DEDICATED venv on the inference server — do NOT reuse the huge
# root requirements.txt (PaddleOCR / spaCy / etc.), which is for the batch
# pipeline only.
#
# Hardware target: NVIDIA TITAN X (Pascal), CC 6.1, driver 555.42.02 (CUDA 12.5).
#
# ── torch + torchvision: install FIRST, from the CUDA 12.4 index ─────────────
#   pip install --index-url https://download.pytorch.org/whl/cu124 \
#       torch==2.6.0 torchvision==0.21.0
# Why: the default PyPI torch wheel is now a CUDA-13 build (`+cu130`), which the
# 555 driver (CUDA 12.5) rejects ("driver too old, found 12050"). The cu124
# build reports arch_list sm_50/sm_60/... — no sm_61, but sm_60 cubins are
# binary-compatible with the sm_61 Pascal cards (same major arch), verified
# working. torchvision is required by the Qwen3-VL image processor. Do NOT let
# `pip install -r requirements.txt` pull torch from PyPI.
#
# The versions below are the exact ones verified by scripts/inference_app_
# smoketest.py on this hardware.
# -----------------------------------------------------------------------------

torch==2.6.0               # install from the cu124 index first (see note above)
torchvision==0.21.0        # cu124 index too; needed by the Qwen3-VL processor
transformers==5.3.0        # provides the Qwen3-VL classes (matches batch pipeline)
bitsandbytes==0.49.2       # NF4 4-bit; verified on CC 6.1 Pascal
accelerate==1.14.0
faiss-cpu==1.14.3
openai==2.44.0
qwen-vl-utils==0.0.14
streamlit==1.58.0
pillow
numpy
pymupdf                    # read the source PDF page text for the highlight locator
rapidfuzz                  # Levenshtein-based fuzzy match of the quote onto the PDF text
